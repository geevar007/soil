Google Sheets Integration Setup

Google Apps Script ബാക്ക്എൻഡ് സെറ്റ് ചെയ്യാനുള്ള ഗൈഡ്

മലയാളം നിർദ്ദേശങ്ങൾ (Steps in Malayalam):

  1. ഒരു പുതിയ Google Sheet ഉണ്ടാക്കുക.
  2. ആദ്യ വരിയിൽ (Row 1) തലക്കെട്ടുകൾ നൽക്കുക: Block Name (A1), Number (B1), Status (C1).
  3. മുകളിൽ Extensions > Apps Script ഓപ്പൺ ചെയ്യുക.
  4. താഴെ കാണുന്ന Code.gs പൂർണ്ണമായും കോപ്പി ചെയ്ത് Apps Script-ൽ പേസ്റ്റ് ചെയ്യുക.
  5. മുകളിൽ വലതുവശത്തുള്ള Deploy > New deployment ക്ലിക്ക് ചെയ്യുക.
  6. Select type എന്നതിൽ Web App തിരഞ്ഞെടുക്കുക.
  7. Who has access: എന്നത് "Anyone" എന്ന് നിർബന്ധമായും മാറ്റുക.
  8. Deploy അമർത്തുമ്പോൾ കിട്ടുന്ന Web App URL കോപ്പി ചെയ്ത് ഈ വെബ്‌സൈറ്റിലെ API URL ബോക്‌സിൽ പേസ്റ്റ് ചെയ്യുക!
Code.gs (Google Apps Script Code) 
// Google Apps Script Code for Google Sheet
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var rows = [];
  
  // Skip Header Row
  for (var i = 1; i < data.length; i++) {
    rows.push({
      id: i + 1, // Row index in Google Sheet
      blockName: data[i][0] || "",
      number: data[i][1] || "",
      status: data[i][2] || ""
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success", data: rows }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;
    
    if (action === "CREATE") {
      sheet.appendRow([contents.blockName, contents.number, contents.status]);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Added successfully" }))
        .setMimeType(ContentService.MimeType.JSON);
    } 
    
    else if (action === "UPDATE") {
      var rowIndex = parseInt(contents.id);
      sheet.getRange(rowIndex, 1).setValue(contents.blockName);
      sheet.getRange(rowIndex, 2).setValue(contents.number);
      sheet.getRange(rowIndex, 3).setValue(contents.status);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Updated successfully" }))
        .setMimeType(ContentService.MimeType.JSON);
    } 
    
    else if (action === "DELETE") {
      var rowIndex = parseInt(contents.id);
      sheet.deleteRow(rowIndex);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Deleted successfully" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
